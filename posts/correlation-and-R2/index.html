// Function to compute sample correlation (r) from generated points
function getSampleR(points) {
  const n = points.length;
  const meanX = points.reduce((acc, p) => acc + p.x, 0) / n;
  const meanY = points.reduce((acc, p) => acc + p.y, 0) / n;
  
  let num = 0, denX = 0, denY = 0;
  for (let p of points) {
    const dx = p.x - meanX;
    const dy = p.y - meanY;
    num += dx * dy;
    denX += dx * dx;
    denY += dy * dy;
  }
  return num / Math.sqrt(denX * denY);
}

// 2. Generate sample data for the scatter plot based on population r
data = {
  resample; // Triggers evaluation on click
  
  const n = sampleSize;
  const points = [];
  for (let i = 0; i < n; i++) {
    const u1 = Math.random();
    const u2 = Math.random();
    const z0 = Math.sqrt(-2.0 * Math.log(u1)) * Math.cos(2.0 * Math.PI * u2);
    const z1 = Math.sqrt(-2.0 * Math.log(u1)) * Math.sin(2.0 * Math.PI * u2);
    
    // Correlate the variables: Y = r*X + sqrt(1 - r^2)*Z
    const x = z0;
    const y = r * z0 + Math.sqrt(1 - Math.pow(r, 2)) * z1;
    points.push({ x: x, y: y });
  }
  return points;
}

meanX = data.reduce((s, p) => s + p.x, 0) / data.length
meanY = data.reduce((s, p) => s + p.y, 0) / data.length

covXY = data.reduce((s, p) => s + (p.x - meanX) * (p.y - meanY), 0)
varX = data.reduce((s, p) => s + Math.pow(p.x - meanX, 2), 0)

slope = covXY / varX
intercept = meanY - slope * meanX

// Fitted values
dataWithFit = data.map((p, i) => ({
  ...p,
  fitted: intercept + slope * p.x,
  index: i
}))

// Sum-of-squares components
SST = dataWithFit.reduce(
  (s, p) => s + Math.pow(p.y - meanY, 2), 0
)

SSE = dataWithFit.reduce(
  (s, p) => s + Math.pow(p.y - p.fitted, 2), 0
)

SSR = dataWithFit.reduce(
  (s, p) => s + Math.pow(p.fitted - meanY, 2), 0
)

R2_from_SS = SSR / SST

// Choose an observation to illustrate the decomposition

selectedTotal = selectedPoint.y - meanY
selectedResidual = selectedPoint.y - selectedPoint.fitted
selectedExplained = selectedPoint.fitted - meanY

// A square representing a squared deviation.
// The x dimension is deliberately made equal to the y dimension,
// since x and y are both standard-normal scale in this simulation.
function makeSquare(x, y, side) {
  const s = Math.abs(side)
  return [
    { x: x,     y: y },
    { x: x + s, y: y },
    { x: x + s, y: y + (side >= 0 ? s : -s) },
    { x: x,     y: y + (side >= 0 ? s : -s) },
    { x: x,     y: y }
  ]
}

totalSquare = makeSquare(
  selectedPoint.x,
  Math.min(selectedPoint.y, meanY),
  selectedTotal
)

residualSquare = makeSquare(
  selectedPoint.x + 0.15,
  Math.min(selectedPoint.y, selectedPoint.fitted),
  selectedResidual
